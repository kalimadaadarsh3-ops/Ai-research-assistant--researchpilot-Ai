        "    Args:\n",

        "    \"\"\"\n",
        "    results = []\n",
        "    try:\n",
        "        with DDGS() as ddgs:\n",

        "                results.append({\n",
        "                    \"title\": item.get(\"title\", \"\"),\n",
        "                    \"snippet\": item.get(\"body\", \"\"),\n",
        "                    \"url\": item.get(\"href\", \"\")\n",
        "                })\n",
